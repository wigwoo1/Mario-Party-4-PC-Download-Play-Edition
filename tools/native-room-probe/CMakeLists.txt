cmake_minimum_required(VERSION 3.25)
project(partyboard_native_room_probe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe validates the Windows native room HTTP adapter")
endif()
if(NOT EXISTS "${PARTYBOARD_JSON_INCLUDE}/nlohmann/json.hpp")
    message(FATAL_ERROR "PARTYBOARD_JSON_INCLUDE must point to the existing pinned nlohmann include directory")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/NativeRoom.cmake")
add_executable(partyboard_native_room_probe "${CMAKE_CURRENT_LIST_DIR}/../native-room-probe.cpp")
target_include_directories(partyboard_native_room_probe PRIVATE "${PARTYBOARD_JSON_INCLUDE}")
target_link_libraries(partyboard_native_room_probe PRIVATE partyboard_room_http bcrypt)
